# 1.TCP代码
	(1)单次接发收
	`tcp_s.py`

	```
	import socket
	#创建套接字
	sk=socket.socket(socket.AF_INET,socket.SOCK_STREAM)
	#绑定地址
	sk.bind(('127.0.0.1',8080))
	#设置监听
	sk.listen(5)
	#阻塞等待客户端请求
	connsk,addr=sk.accept()
	print('客户端地址：',addr)
	#收发数据
	data=connsk.recv(1024)
	print('接受到的数据',data.decode('utf-8'))
	connsk.send(b'ok')
	#关闭套接字
	connsk.close()
	sk.close()
	```

# 3.聊天室
udp_server.py
```
import socket

HOST = '127.0.0.1'
PORT = 9000

def main():

    # 创建 UDP 套接字

    server_socket = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)

    server_socket.bind((HOST, PORT))

    print(f"🚀 UDP 聊天室服务端启动！监听 {HOST}:{PORT}")



    try:
        while True:
            # recvfrom 接收数据和客户端地址（UDP 特有）
            data, addr = server_socket.recvfrom(1024)
            msg = f"[UDP {addr[0]}:{addr[1]}] {data.decode('utf-8')}"
            print("收到 →", msg)

            # 广播给所有（这里简单处理：只广播给当前客户端和部分历史地址）
            # 实际可维护 clients 列表（类似 TCP）

            server_socket.sendto(msg.encode(), addr)  # 回个确认

            # 也可广播给其他客户端（需提前记录）
            # server_socket.sendto(msg.encode(), ('<broadcast>', PORT))  # 需设置 SO_BROADCAST

    except KeyboardInterrupt:
        print("\n👋 服务端关闭")
    finally:
        server_socket.close()

if __name__ == "__main__":

    main()
```
udp_client.py
```
import socket

import threading



SERVER_HOST = '127.0.0.1'

SERVER_PORT = 9000



def receive_udp(sock):

    """UDP 接收线程"""

    while True:

        try:

            data, addr = sock.recvfrom(1024)

            print("\n[UDP 收到]", data.decode())

        except Exception as e:

            print("接收出错:", e)

            break



def main():

    sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)

    # 可选：设置广播权限（UDP 特有）

    # sock.setsockopt(socket.SOL_SOCKET, socket.SO_BROADCAST, 1)

    print("✅ UDP 客户端就绪，输入消息发送给服务端：")



    # 启动接收线程

    recv_thread = threading.Thread(target=receive_udp, args=(sock,), daemon=True)

    recv_thread.start()



    try:

        while True:

            msg = input().strip()

            if msg.lower() in ['quit', 'exit', 'q']:

                break

            sock.sendto(msg.encode(), (SERVER_HOST, SERVER_PORT))

    except KeyboardInterrupt:

        print("\n👋 断开连接")

    finally:

        sock.close()



if __name__ == "__main__":

    main()
```
# 4.TCP协议的各阶段编程

## 服务器循环接收 + 回显功能
tcp_s.py
```
import socket
#创建套接字
sk=socket.socket(socket.AF_INET,socket.SOCK_STREAM)
#绑定地址
sk.bind(('127.0.0.1',8081))
#设置监听
sk.listen(5)
print("服务端已启动，等待客户连接")
#阻塞等待客户端请求
connsk,addr=sk.accept()
print(f'客户端{addr}已连接')
#收发数据
try:
    while True:
        data=connsk.recv(1024)
        if not data:
            print("客户端关闭连接")
            break
        msg=data.decode('utf-8').strip()
        #====[新增]:回显功能====
        if msg.upper()=='QUIT':
            reply='Goodebye'
        else:
            reply=f'{msg}'#服务器原样回显
        connsk.send(reply.encode('utf-8'))
        print(f'回显给客户端:{repr(reply)}')
# print('接受到的数据',data.decode('utf-8'))
# connsk.send(b'data')
#关闭套接字
finally:
    connsk.close()
    sk.close()
    print("服务端退出")
```
tcp_c.py
```
import socket

sk=socket.socket(socket.AF_INET,socket.SOCK_STREAM)

#连接服务器，下面括号里面的东西是一个元组

sk.connect(('127.0.0.1',8081))
print("已连接，输入quit退出")
#循环收发数据
while True:
    try:
        msg=input("你:").strip()
        if not msg:
            continue
        #编辑成字节流
        sk.send(msg.encode('utf-8'))
        reply=sk.recv(1024).decode('utf-8').strip()
        print(f"服务端:{reply}")
        if reply=='Goodebye':
            break
    except:
        sk.send(b'QUIT')
        break
#关闭套接字
sk.close()
print("客户端退出")
```
